# covid_uptake siblings
cov_prepped_ivd |>
  
  filter(!is.na(covid_coverage_age)) |>
  
  # filter out partial uptake
  filter(covid_coverage_age != 1) |>
  
  # rename sibling categories and compress them
  mutate(
     child_siblings = case_when(
      child_siblings == 0 ~ "No Siblings",
      child_siblings == 1 ~ "1 Sibling",
      child_siblings >= 2 ~ "2+ Siblings"
      ) |>
       factor(
         levels= c("No Siblings", "1 Sibling", "2+ Siblings")
       ),
    
    # segment age into increments of 0.5 years
    age_of_interest = round(age_of_interest * 2) / 2
  ) |>
  
  summarise(count = n(), .by = c(age_of_interest, covid_coverage_age, child_siblings, age_cat)) |>
  mutate(proportion = count / sum (count), .by = c(age_of_interest, child_siblings, age_cat)) |>
  arrange(child_siblings, age_of_interest, covid_coverage_age) |>
  
  mutate(
    # weight observations within each age_category, such that the trend for the proportion that have received each coverage is reflective of the sample size
    weights = count / sum(count), .by = c(age_cat, child_siblings, covid_coverage_age)
    
  ) |>
  
  # remove outliers where proportion = 1
  filter(proportion != 1) |>
  
  # rename coverage, number of siblings and schooling statuses
  mutate(
    covid_coverage_age = case_when(
      covid_coverage_age == 0 ~ "Unvaccinated",
      covid_coverage_age == 2 ~ "Fully Vaccinated"
    ),
    
    age_cat = case_when(
      age_cat == "pre" ~ "Preschool",
      age_cat == "primary" ~ "Primary school",
      age_cat == "high" ~ "(Junior) Highschool"
    )
  ) |>
   
  # plot graphs of covid coverage by age across different birth orders
  ggplot(aes(x = age_of_interest, y = proportion, colour = covid_coverage_age)) +
  geom_point(aes(shape = age_cat)) +
  geom_smooth(aes(linetype = age_cat, weight = weights), size = 0.75, alpha = 0.1, method = "lm") +
  facet_wrap(vars(child_siblings), ncol = 3) +
  labs(
    title = "Covid vaccination status with children's age,\n number of siblings and current level of schooling",
    x = "Age",
    y = "Proportion of children",
    col = "Covid vaccination status",
    shape = "Schooling category",
    linetype = "Schooling category"
  ) +
  
  theme(
    text = element_text(family = "sans"),
    plot.title = element_text(size = 16, face = "bold", margin = margin(l = 15, b = 10)),
    plot.title.position = "plot",
    axis.title.x = element_text(size = 12),
    axis.title.y = element_text(size = 12),
    legend.position = "bottom",
    legend.box = "vertical", # If you have multiple legends, arrange them vertically
    #legend.box.just = "top", # Align to the top of the legend box area
    #legend.justification = "top", # Justify the legend key within its area
    legend.text = element_text(lineheight = 0.5),
    legend.margin = margin(t = -10, r = 0, b = 0, l = 0),
    panel.grid.major = element_line(colour = "grey95"),
    panel.grid.minor = element_line(colour = "grey95"),
    panel.background = element_rect(fill = "white"),
    panel.border = element_rect(fill = "NA",colour = "grey90")
  ) +
  
  guides(
    color = guide_legend(override.aes = list(fill = NA)),
    shape = guide_legend(override.aes = list(fill = NA)), 
    linetype = guide_legend(override.aes = list(fill = NA)) 
  )
